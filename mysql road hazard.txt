CREATE road_hazard_db;
USE road_hazard_db;
-- TABLE 1: roads
CREATE TABLE roads (
    road_id INT PRIMARY KEY AUTO_INCREMENT,
    road_name VARCHAR(100) NOT NULL
);
-- TABLE 2: hazards 
CREATE TABLE hazards (
    hazard_id INT PRIMARY KEY AUTO_INCREMENT,
    road_id INT,
    hazard_type VARCHAR(50),
    FOREIGN KEY (road_id) REFERENCES roads(road_id) ON DELETE CASCADE
);
-- TABLE 3: severity
CREATE TABLE severity (
    severity_id INT PRIMARY KEY AUTO_INCREMENT,
    road_id INT,
    severity VARCHAR(20),
    FOREIGN KEY (road_id) REFERENCES roads(road_id) ON DELETE CASCADE
);

-- Tables output
SELECT * FROM roads;
SELECT * FROM hazards;
SELECT * FROM severity;
-- Join of ALL THREE tables
SELECT r.road_id,
       r.road_name,
       h.hazard_type,
       s.severity
FROM roads r
LEFT JOIN hazards  h ON r.road_id = h.road_id
LEFT JOIN severity s ON r.road_id = s.road_id
ORDER BY r.road_id, h.hazard_id;
-- Summary: one row per road
SELECT r.road_name,
       COUNT(DISTINCT h.hazard_id) AS total_hazards,
       MAX(s.severity)             AS road_severity
FROM roads r
LEFT JOIN hazards  h ON r.road_id = h.road_id
LEFT JOIN severity s ON r.road_id = s.road_id
GROUP BY r.road_id, r.road_name
ORDER BY r.road_id;
